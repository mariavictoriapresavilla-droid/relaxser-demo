/** * Relaxser · Reservas conectadas a Google Calendar * * Qué hace: * 1) La página consulta este programa para saber qué horarios están ocupados. * Un horario está ocupado si en su Google Calendar hay cualquier evento * (reserva, evento de todo el día, turno médico, etc.) que se superponga. * 2) Cuando un cliente toca "Solicitar reserva", el programa crea un evento * "pendiente" (naranja) en su calendario, así nadie más puede tomar ese horario. * 3) Usted confirma por WhatsApp. Si NO se confirma, borre el evento y el * horario vuelve a quedar libre en la página. * * Los horarios de atención se cambian acá abajo, en SCHEDULE. */ const CONFIG = { CALENDAR_ID: '', // vacío = su calendario principal. Para usar otro, pegue su ID entre las comillas. TZ: 'America/Montevideo', // Uruguay (UTC-3) HORIZON_DAYS: 90, // hasta cuántos días adelante se puede reservar (igual que la página) LEAD_MIN: 120, // anticipación mínima: 2 horas (igual que la página) SLOT_MIN: 90, // minutos que se bloquean por reserva (servicio + traslado) MAX_PER_HOUR: 12, // tope de reservas nuevas por hora (evita abusos / spam) NOTIFY_EMAIL: true // le envía un mail cuando entra una solicitud }; // Horarios de atención por día de la semana (0 = domingo ... 6 = sábado). // Cada horario bloquea SLOT_MIN minutos. const WK = ['15:00', '16:30', '18:00']; const SA = ['09:00', '10:30', '14:00', '15:30', '17:00', '18:30']; const SCHEDULE = { 0: [], 1: WK, 2: WK, 3: WK, 4: WK, 5: WK, 6: SA }; /* ---------- Entrada web ---------- */ function doGet(e) { const p = (e && e.parameter) || {}; let out; try { out = p.action === 'book' ? book_(p) : { ok: true, busy: busy_(), schedule: SCHEDULE }; } catch (err) { console.error(err); out = { ok: false, error: 'server' }; } return ContentService.createTextOutput(JSON.stringify(out)) .setMimeType(ContentService.MimeType.JSON); } /* ---------- Utilidades ---------- */ function cal_() { const c = CONFIG.CALENDAR_ID ? CalendarApp.getCalendarById(CONFIG.CALENDAR_ID) : CalendarApp.getDefaultCalendar(); if (!c) throw new Error('No se encontró el calendario'); return c; } function fmt_(d, f) { return Utilities.formatDate(d, CONFIG.TZ, f); } function dow_(d) { return Number(fmt_(d, 'u')) % 7; } // 0 = domingo function slotDate_(day, time) { return Utilities.parseDate(day + ' ' + time, CONFIG.TZ, 'yyyy-MM-dd HH:mm'); } function clean_(v, n) { return String(v || '').replace(/[\r\n\t]+/g, ' ').trim().slice(0, n); } // Todos los horarios posibles de los próximos HORIZON_DAYS días. function candidateSlots_() { const out = [], now = Date.now(); for (let i = 0; i <= CONFIG.HORIZON_DAYS; i++) { const d = new Date(now + i * 864e5); const day = fmt_(d, 'yyyy-MM-dd'); (SCHEDULE[dow_(d)] || []).forEach(function (t) { const s = slotDate_(day, t); out.push({ key: day + ' ' + t, start: s, end: new Date(s.getTime() + CONFIG.SLOT_MIN * 60000) }); }); } return out; } // Horarios ocupados: los que se superponen con algún evento del calendario. function busy_() { const slots = candidateSlots_(); if (!slots.length) return []; const from = slots[0].start, to = slots[slots.length - 1].end; const spans = cal_().getEvents(from, to).map(function (e) { return [e.getStartTime().getTime(), e.getEndTime().getTime()]; }); return slots.filter(function (s) { return spans.some(function (a) { return a[0] < s.end.getTime() && a[1] > s.start.getTime(); }); }).map(function (s) { return s.key; }); } /* ---------- Reservar ---------- */ function book_(p) { const m = /^(\d{4}-\d{2}-\d{2}) (\d{2}:\d{2})$/.exec(String(p.slot || '')); if (!m) return { ok: false, error: 'invalid' }; const start = slotDate_(m[1], m[2]); if ((SCHEDULE[dow_(start)] || []).indexOf(m[2]) < 0) return { ok: false, error: 'invalid' }; const ahead = start.getTime() - Date.now(); if (ahead < CONFIG.LEAD_MIN * 60000 || ahead > (CONFIG.HORIZON_DAYS + 1) * 864e5) { return { ok: false, error: 'invalid' }; } const name = clean_(p.name, 80), address = clean_(p.address, 160); const notes = clean_(p.notes, 400), service = clean_(p.service, 80) || 'Masaje'; if (!name || !address) return { ok: false, error: 'invalid' }; const lock = LockService.getScriptLock(); if (!lock.tryLock(15000)) return { ok: false, error: 'busy' }; try { const cache = CacheService.getScriptCache(); const n = Number(cache.get('rl') || 0); if (n >= CONFIG.MAX_PER_HOUR) return { ok: false, error: 'limit' }; const end = new Date(start.getTime() + CONFIG.SLOT_MIN * 60000); const cal = cal_(); if (cal.getEvents(start, end).length) return { ok: false, error: 'taken' }; const when = fmt_(start, 'dd/MM/yyyy HH:mm'); const desc = 'Solicitud hecha desde la página web.\n\n' + 'Cliente: ' + name + '\n' + 'Servicio: ' + service + '\n' + 'Dirección: ' + address + '\n' + (notes ? 'Observaciones: ' + notes + '\n' : '') + '\nPENDIENTE de confirmar por WhatsApp. ' + 'Si no se confirma, borre este evento para liberar el horario.'; const ev = cal.createEvent('⏳ ' + service + ' · ' + name + ' (pendiente)', start, end, { location: address, description: desc }); try { ev.setColor(CalendarApp.EventColor.ORANGE); } catch (e) { /* el color es opcional */ } cache.put('rl', String(n + 1), 3600); if (CONFIG.NOTIFY_EMAIL) { try { MailApp.sendEmail(Session.getEffectiveUser().getEmail(), 'Nueva solicitud de reserva: ' + when, desc + '\n\nFecha y hora: ' + when); } catch (e) { console.error(e); } } return { ok: true }; } finally { lock.releaseLock(); } } /* ---------- Se ejecuta UNA sola vez a mano, para dar los permisos ---------- */ function autorizar() { Logger.log('Calendario conectado: ' + cal_().getName()); Logger.log('Mails disponibles hoy: ' + MailApp.getRemainingDailyQuota()); Logger.log('Horarios ocupados próximos 90 días: ' + busy_().length); }
